# 필름배지 콜로니 카운터 통합 도구 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `Petricore_Colony_Counter_최종.html`을 기반으로, v2.2의 OCR·해시체인 원장·영속화·유채색 잉크마스크·분리모드·계수 프리셋을 이식한 단일 파일 도구 `필름배지/index.html`를 만든다.

**Architecture:** 최종 파일을 그대로 `필름배지/index.html`로 복사한 뒤 증분 수정한다. 최종의 `/* ===== js/xxx.js ===== */` 구획 관례를 유지하고, `js/ledger.js` · `js/inkMask.js` · `js/persistence.js` 구획을 신규 추가, `js/ocr.js` 구획을 전면 교체한다. 캔버스에 묶인 함수는 "순수 코어 + 캔버스 래퍼"로 분리해 Node에서 단위 테스트한다. 빌드 도구·npm 의존성 없음 — 테스트는 `index.html`에서 구획을 잘라내 `node:vm`으로 평가한다.

**Tech Stack:** 순수 브라우저 JS(ES2020), 인라인 CSS. 외부 라이브러리(핀 고정): SheetJS xlsx 0.18.5, PDF.js 3.11.174, Tesseract.js 5, JSZip 3.10.1. 테스트: Node `node:test` + `node:vm` (npm 설치 없음).

**Spec:** `docs/superpowers/specs/2026-09-11-필름배지-콜로니카운터-design.md`

## Global Constraints

- 단일 HTML 파일. 빌드 단계 없음. `file://` 더블클릭으로 동작해야 한다.
- 외부 라이브러리는 정확히 이 버전/URL만 사용: xlsx `https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js`, pdf.js `https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js`, pdf.worker `https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js`, tesseract.js `https://cdn.jsdelivr.net/npm/tesseract.js@5/dist/tesseract.min.js`, jszip `https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js`.
- localStorage 키: `petricore.film.v1` (데이터), `petricore.film.env` (환경 점검 결과), `petricore-theme` (기존 유지).
- 원장: `GEN = '0'.repeat(32)`, 필드 구분자 `SEP = ''`. 이벤트에 `actor`/서명 필드 없음.
- 저장 좌표는 소수 5자리 반올림. 원장 `pts`는 소수 4자리.
- 내보낸 파일에 제로폭 지문 워터마크를 넣지 않는다.
- 균종 동정(species identification)을 시사하는 UI 문구 금지.
- 모든 사용자 대면 문구는 한국어.
- 최종 파일의 기존 화면·기능(메인 표, 읽기전용 뷰어, 콜로니 편집 모달, Excel 내보내기, 이미지 ZIP, 다크모드, 검색/Type 필터/페이지네이션, PDF 입력)은 유지한다.

// </script>) 직전까지를 잘라낸다.
function sliceSection(html, name) {
  const marker = `/* ===== js/${name} ===== */`;
  const start = html.indexOf(marker);
  if (start < 0) throw new Error(`section not found: ${name}`);
  const rest = html.slice(start + marker.length);
  const nextMarker = rest.search(/\/\* ===== js\/[\w.]+ ===== \*\//);
  const endScript = rest.indexOf('</script>');
  let end = rest.length;
  if (nextMarker >= 0) end = nextMarker;
  if (endScript >= 0 && endScript < end) end = endScript;
  return rest.slice(0, end);
}

export function makeStubContext() {
  const store = new Map();
  const localStorage = {
    getItem: k => (store.has(k) ? store.get(k) : null),
    setItem: (k, v) => store.set(k, String(v)),
    removeItem: k => store.delete(k),
  };
  const noopCanvas = () => ({
    width: 0, height: 0,
    getContext: () => ({
      drawImage() {}, fillRect() {}, clearRect() {},
      getImageData: (x, y, w, h) => ({ data: new Uint8ClampedArray(w * h * 4), width: w, height: h }),
      putImageData() {}, translate() {}, rotate() {},
    }),
  });
  const document = {
    createElement: tag => (tag === 'canvas' ? noopCanvas() : { style: {}, appendChild() {}, addEventListener() {} }),
    getElementById: () => null,
    querySelector: () => null,
    addEventListener() {},
  };
  const ctx = {
    console, performance: { now: () => 0 },
    document, localStorage,
    window: { localStorage, matchMedia: () => ({ matches: false }), addEventListener() {} },
    TextEncoder, TextDecoder, Uint8Array, Uint8ClampedArray, Int32Array, Float32Array, Float64Array,
    Math, JSON, Date, BigInt, Object, Array,
  };
  ctx.globalThis = ctx;
  return vm.createContext(ctx);
}

export default function loadSections(sectionNames) {
  const html = readIndex();
  const src = sectionNames.map(n => sliceSection(html, n)).join('\n;\n');
  const wanted = [...new Set(sectionNames.flatMap(n => EXPORTS[n] || []))];

- [ ] **Step 3: `js/ledger.js` 구획 삽입**

`필름배지/index.html`의 `/* ===== js/detector.js ===== */` 마커 **바로 앞** 줄에 아래 블록을 삽입:

```js
/* ===== js/ledger.js ===== */
/**
 * 해시체인 이벤트 로그. v2.2에서 이식하되 작업자/서명 필드는 제거했다.
 * 모든 데이터 변경(콜로니 편집·메타 수정·삭제·OCR·내보내기)은 ledger.commit()을
 * 거쳐 append-only 로 기록되고, verifyChain()으로 무결성을 재검증할 수 있다.
 * per-record undo(historyPast/historyFuture)와는 별개이며 서로 간섭하지 않는다.
 */
const GEN = '0'.repeat(32);
const SEP = '';

function h128(s) {
  const P = 1099511628211n, M = 0xffffffffffffffffn;
  const b = new TextEncoder().encode(String(s));
  let a = 14695981039346656037n, c = 11400714819323198485n;
  for (let i = 0; i < b.length; i++) {
    const x = BigInt(b[i]);
    a = ((a ^ x) * P) & M;
    c = ((c ^ (x + 7n)) * P) & M;
    c = ((c << 13n) | (c >> 51n)) & M;
  }
  return a.toString(16).padStart(16, '0') + c.toString(16).padStart(16, '0');
}

function hashOf(e) {
  return h128([e.prev, e.seq, e.ts, e.kind, JSON.stringify(e.payload)].join(SEP));
}

function pushEvent(kind, payload, list) {
  const l = (list || state.events || []).slice();
  const prev = l.length ? l[l.length - 1].hash : GEN;
  const ev = { seq: l.length + 1, ts: new Date().toISOString(), kind, payload, prev };
  ev.hash = hashOf(ev);
  l.push(ev);
  return l;
}

function verifyChain(events) {
  let prev = GEN;
  for (let i = 0; i < events.length; i++) {
    const e = events[i];
    if (e.prev !== prev || e.seq !== i + 1 || hashOf(e) !== e.hash) return { ok: false, at: i + 1 };
    prev = e.hash;
  }
  return { ok: true, at: 0 };
}

const ledger = {
  commit(kind, payload) {
    state.events = pushEvent(kind, payload);
    if (typeof saveState === 'function') saveState();
  },
  tip(list) {
    const l = list || state.events;
    return l.length ? l[l.length - 1].hash : GEN;
  },
  exportJSON() {
    return JSON.stringify({ v: 1, exportedAt: new Date().toISOString(), events: state.events }, null, 2);
  },
  importJSON(text) {
    let parsed;
    try { parsed = JSON.parse(text); } catch (e) { return { ok: false, at: 0, events: [] }; }
    const events = Array.isArray(parsed) ? parsed : (parsed.events || []);
    const r = verifyChain(events);
    return { ok: r.ok, at: r.at, events };
  },
};

/* ===== js/persistence.js ===== */
/**
 * localStorage 영속화. 원본 캔버스 대신 썸네일(dataURL)과 정규화 좌표만 저장한다.
 * 복원 시 캔버스는 썸네일 해상도이며 record.lowRes=true 로 표시된다(재검출 정확도 제한).
 * 원장(state.events)이 진실의 원천에 가깝고, 가져오기 시에는 이벤트 재생으로
 * 레코드를 재구성한다(rebuildFromEvents).

const LS_KEY = 'petricore.film.v1';
const R5 = n => Math.round(n * 1e5) / 1e5;

function normalizePts(colonies, w, h) {
  return (colonies || []).map(c => [
    R5(c.x / w), R5(c.y / h), R5((c.radius || 0) / w), c.color === 'blue' ? 1 : 0,
  ]);
}
function denormalizePts(arr, w, h) {
  return (arr || []).map(a => ({
    x: a[0] * w, y: a[1] * h, radius: (a[2] || 0) * w, color: a[3] === 1 ? 'blue' : 'red',
  }));
}

function thumbOf(canvas, maxSide) {
  maxSide = maxSide || 600;
  const s = Math.min(1, maxSide / Math.max(canvas.width, canvas.height));
  const o = document.createElement('canvas');
  o.width = Math.max(1, Math.round(canvas.width * s));
  o.height = Math.max(1, Math.round(canvas.height * s));
  o.getContext('2d').drawImage(canvas, 0, 0, o.width, o.height);
  return o.toDataURL('image/jpeg', 0.76);
}
function thumbToCanvas(url) {
  return new Promise(res => {
    const img = new Image();
    img.onload = () => {
      const c = document.createElement('canvas');
      c.width = img.naturalWidth; c.height = img.naturalHeight;
      c.getContext('2d').drawImage(img, 0, 0);
      res(c);
    };
    img.onerror = () => res(document.createElement('canvas'));
    img.src = url;
  });
}

function serializeState() {
  const records = state.records.map(r => ({
    id: r.id, filename: r.filename,
    type: r.type, lot: r.lot, bacteria: r.bacteria, no: r.no, dilute: r.dilute,
    rotation: r.rotation || 0,
    sensitivity: r.sensitivity, minArea: r.minArea, colorMode: r.colorMode,
    labelHeightRatio: r.labelHeightRatio,
    imgW: r.canvas.width, imgH: r.canvas.height,
    thumb: thumbOf(r.canvas, 600),
    colonies: normalizePts(r.colonies, r.canvas.width, r.canvas.height),
    needsReview: r.needsReview || {},
    ocrSource: r.ocrSource || {},
    rawHandwriting: r.rawHandwriting || '', rawPrinted: r.rawPrinted || '',
  }));
  return {
    v: 1, savedAt: new Date().toISOString(),
    splitMode: state.splitMode, rangePreset: state.rangePreset,
    records, events: state.events,
  };
}

function hydrateState(raw) {
  if (!raw || !raw.records) return;
  state.splitMode = raw.splitMode || 'normal';
  state.rangePreset = raw.rangePreset || 'aoac';
  state.events = raw.events || [];
  const pending = raw.records.map(async sr => {
    const canvas = await thumbToCanvas(sr.thumb);
    return {
      id: sr.id, filename: sr.filename, canvas, lowRes: true,
      imgW: sr.imgW, imgH: sr.imgH,
      rotation: sr.rotation || 0,
      type: sr.type, lot: sr.lot, bacteria: sr.bacteria, no: sr.no, dilute: sr.dilute,
      sensitivity: sr.sensitivity, minArea: sr.minArea, colorMode: sr.colorMode,
      labelHeightRatio: sr.labelHeightRatio,
      colonies: denormalizePts(sr.colonies, canvas.width, canvas.height),
      autoColonies: [],
      historyPast: [], historyFuture: [],
      selected: false,
      needsReview: sr.needsReview || {},
      ocrSource: sr.ocrSource || {},
      rawHandwriting: sr.rawHandwriting || '', rawPrinted: sr.rawPrinted || '',
    };
  });
  return Promise.all(pending).then(recs => {
    state.records = recs;
    recs.forEach(r => { r.autoColonies = JSON.parse(JSON.stringify(r.colonies)); });
  });
}

function saveState() {
  try {
    localStorage.setItem(LS_KEY, JSON.stringify(serializeState()));
  } catch (e) {
    if (typeof toast === 'function') toast('저장 공간이 부족합니다 · 원장을 내보내 주세요', 'warn');
  }
}

function loadState() {
  let raw = null;
  try { raw = JSON.parse(localStorage.getItem(LS_KEY) || 'null'); } catch (e) { raw = null; }
  if (!raw) return Promise.resolve(false);
  return Promise.resolve(hydrateState(raw)).then(() => true);
}

function rebuildFromEvents(events) {
  const map = new Map();
  for (const e of events) {
    const p = e.payload || {};
    if (e.kind === 'PLATE_ADD') {
      map.set(p.id, {
        id: p.id, filename: p.filename || '',
        type: p.type || '', lot: p.lot || '', bacteria: p.bacteria || '',
        no: p.no || '', dilute: p.dilute || '',
        imgW: p.w, imgH: p.h, thumb: p.thumb || '',
        _pts: p.pts || [],
        needsReview: {}, ocrSource: {},
      });
    } else if (e.kind === 'META_SET' && map.has(p.id)) {
      map.get(p.id)[p.field] = p.to;
    } else if (e.kind === 'COUNT_SET' && map.has(p.id)) {
      map.get(p.id)._pts = p.pts || map.get(p.id)._pts;
    } else if (e.kind === 'OCR_READ' && map.has(p.id)) {
      Object.assign(map.get(p.id), p.read || {});
    } else if (e.kind === 'PLATE_DEL') {
      map.delete(p.id);
    }
  }
  return [...map.values()].map(r => ({
    ...r,
    colonies: denormalizePts(r._pts, r.imgW || 1, r.imgH || 1),
  }));
}

/* ===== js/inkMask.js ===== */
/**
 * 필름 잉크 마스크 — 유채색(연녹 격자)이거나 어두운 픽셀만 잉크로 본다.
 * 스캐너의 중성 회색 잡티(밝고 무채색)는 걸러진다. v2.2의 inkMask를 이식하되
 * 다운스케일 없이 캔버스 원해상도에서 동작한다(splitPanels의 좌표 계산 무수정).
 */
function inkMaskFromImageData(data, w, h, opts) {
  const whiteThreshold = (opts && opts.whiteThreshold) || 232;
  const chroma = (opts && opts.chroma) || 12;
  const raw = new Uint8Array(w * h);
  for (let p = 0; p < w * h; p++) {
    const i = p * 4, R = data[i], G = data[i + 1], B = data[i + 2];
    const mx = R > G ? (R > B ? R : B) : (G > B ? G : B);
    const mn = R < G ? (R < B ? R : B) : (G < B ? G : B);
    raw[p] = (mx - mn > chroma || mx < whiteThreshold) ? 1 : 0;
  }
  const m = new Uint8Array(w * h);
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      const p = y * w + x;
      if (!raw[p]) continue;
      const n = raw[p - w - 1] + raw[p - w] + raw[p - w + 1] + raw[p - 1] + raw[p]
              + raw[p + 1] + raw[p + w - 1] + raw[p + w] + raw[p + w + 1];
      if (n >= 3) m[p] = 1;
    }
  }
  return m;
}

function buildInkMask(canvas, opts) {
  const w = canvas.width, h = canvas.height;
  const ctx = canvas.getContext('2d', { willReadFrequently: true });
  const { data } = ctx.getImageData(0, 0, w, h);
  return inkMaskFromImageData(data, w, h, opts);
}

/**
 * Tesseract.js 상단 라벨 OCR — v2.2 파이프라인 이식.
 *  - 워커 1개 재사용, 영역마다 여러 PSM을 시도해 점수 최고값 채택
 *  - 전처리: 초록 격자선 제거 → Otsu 이진화 → 2~5배 업스케일
 *  - Levenshtein 퍼지매칭(KNOWN_TYPES/KNOWN_STRAINS), 숫자혼동 교정
 *  - 못 찾은 필드는 빈 문자열로 둔다 (기본값을 지어내지 않음 — 정직한 실패)
 *  - 항상 rawPrinted / rawHandwriting 원문을 함께 반환
 */
const KNOWN_TYPES = ['AC', 'EC', 'YM', 'SA', 'CC', 'AP', 'EB', 'TC', 'RYM', 'STX'];
const KNOWN_STRAINS = ['AC', 'CC', 'EC', 'YM', 'SA', 'BC', 'LM', 'ST', 'EB', 'PA', 'KP', 'SE'];
const OCR_ALNUM = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-/ ';

let _ocrWorker = null;
async function getOcrWorker() {
  if (!_ocrWorker) _ocrWorker = await Tesseract.createWorker('eng');
  return _ocrWorker;
}
async function terminateOCR() {
  if (_ocrWorker) { try { await _ocrWorker.terminate(); } catch (e) {} _ocrWorker = null; }
}

function lev(a, b) {
  const m = a.length, n = b.length;
  if (!m || !n) return Math.max(m, n);
  let prev = Array.from({ length: n + 1 }, (_, i) => i);
  for (let i = 1; i <= m; i++) {
    const cur = [i];
    for (let j = 1; j <= n; j++)
      cur[j] = Math.min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1));
    prev = cur;
  }
  return prev[n];
}
function levClosest(tok, list) {
  if (!tok) return '';
  let best = '', bd = 99;
  list.forEach(c => { const d = lev(tok, c); if (d < bd) { bd = d; best = c; } });
  return bd <= (tok.length <= 2 ? 0 : 1) ? best : '';
}
function fixDigits(s) {
  s = String(s);
  return s.replace(/[OIS]/g, (ch, i) => {
    const prev = s[i - 1], next = s[i + 1];
    const d = c => c && c >= '0' && c <= '9';
    if (d(prev) || d(next)) return ch === 'O' ? '0' : ch === 'I' ? '1' : '5';
    return ch;
  });
}

// 영역 크롭 + 초록 격자선 제거 + Otsu 이진화 + 업스케일. v2.2 prep 이식.
function prepRegion(src, x, y, w, h) {
  const cw = Math.max(1, Math.round(src.width * w)), chh = Math.max(1, Math.round(src.height * h));
  const sc = Math.max(2, Math.min(5, 900 / cw));
  const o = document.createElement('canvas');
  o.width = Math.round(cw * sc); o.height = Math.round(chh * sc);
  const c = o.getContext('2d', { willReadFrequently: true });
  c.imageSmoothingEnabled = true; c.imageSmoothingQuality = 'high';
  c.fillStyle = '#fff'; c.fillRect(0, 0, o.width, o.height);
  c.drawImage(src, Math.round(src.width * x), Math.round(src.height * y), cw, chh, 0, 0, o.width, o.height);
  const d = c.getImageData(0, 0, o.width, o.height), px = d.data;
  const hist = new Array(256).fill(0);
  for (let i = 0; i < px.length; i += 4) {
    const R = px[i], G = px[i + 1], B = px[i + 2];
    const g = (G > R + 8 && G > B + 16) ? 255 : (R * 0.35 + G * 0.35 + B * 0.3) | 0;
    px[i] = px[i + 1] = px[i + 2] = g; hist[g]++;
  }
  const total = o.width * o.height;
  let sum = 0; for (let i = 0; i < 256; i++) sum += i * hist[i];
  let sumB = 0, wB = 0, mx = 0, th = 128;
  for (let i = 0; i < 256; i++) {
    wB += hist[i]; if (!wB) continue;
    const wF = total - wB; if (!wF) break;
    sumB += i * hist[i];
    const mB = sumB / wB, mF = (sum - sumB) / wF, v = wB * wF * (mB - mF) * (mB - mF);
    if (v > mx) { mx = v; th = i; }
  }
  const cut = Math.min(238, th + 14);
  for (let i = 0; i < px.length; i += 4) { const v = px[i] > cut ? 255 : 0; px[i] = px[i + 1] = px[i + 2] = v; }
  c.putImageData(d, 0, 0);
  return o;
}

function parsePrinted(text) {
  const T = String(text || '').toUpperCase().replace(/[^A-Z0-9\n ]/g, ' ');
  const lines = T.split('\n').map(s => s.trim()).filter(Boolean);
  let type = '';
  for (const ln of lines) { const m = ln.match(/PETRI\S*\s+([A-Z]{2,3})(?![A-Z])/); if (m) { type = m[1]; break; } }
  if (!type) {
    outer: for (const ln of lines) for (const tk of ln.split(/\s+/)) {
      const c = levClosest(tk, KNOWN_TYPES); if (c) { type = c; break outer; }
    }
  }
  const cands = [];
  for (const ln of lines) for (const tk of ln.split(/\s+/)) {
    const s = tk.replace(/[^A-Z0-9]/g, '');
    if (s.length >= 6 && /[A-Z]/.test(s) && /\d/.test(s) && !/^PETRI/.test(s)) cands.push(s);
  }
  let lot = cands.length ? cands[cands.length - 1] : '';
  if (lot) lot = fixDigits(lot);
  return { type, lot };
}

function parseHand(text) {
  const T = String(text || '').toUpperCase();
  let bacteria = '', weak = false;
  const toks = T.match(/[A-Z]{2,4}/g) || [];
  for (const tk of toks) { const c = levClosest(tk, KNOWN_STRAINS); if (c) { bacteria = c; break; } }
  if (!bacteria && toks.length) { bacteria = toks[0].slice(0, 3); weak = true; }
  let no = '', dilute = '';
  const pair = T.match(/(\d)\s*[-–—~/]\s*(\d)/);
  if (pair) { no = pair[1]; dilute = pair[2]; }
  else {
    const ds = T.match(/\d/g) || [];
    if (ds.length >= 2) { no = ds[0]; dilute = ds[ds.length - 1]; }
    else if (ds.length === 1) dilute = ds[0];
  }
  return { bacteria, no, dilute, weak };
}

async function parsePetrifilmOCR(panelCanvas, labelHeightRatio = 0.12) {
  if (typeof Tesseract === 'undefined') {
    return { type: '', lot: '', bacteria: '', no: '', dilute: '',
      weak: { bacteria: false }, rawHandwriting: '(OCR 없음 · 파일명/직전값 추정)', rawPrinted: '' };
  }
  try {
    const w = await getOcrWorker();
    const band = Math.max(0.16, labelHeightRatio + 0.04);
    const read = async (cv, psms) => {
      const out = [];
      for (const psm of psms) {
        await w.setParameters({ tessedit_char_whitelist: OCR_ALNUM, tessedit_pageseg_mode: psm });
        const r = await w.recognize(cv);
        out.push((r.data.text || '').trim());
      }
      return out;
    };
    let bp = { type: '', lot: '', raw: '' }, bps = -1;
    for (const tx of await read(prepRegion(panelCanvas, 0.42, 0, 0.58, band), ['6', '4'])) {
      const p = parsePrinted(tx);
      const s = (p.type ? 1 : 0) + (p.lot ? 2 : 0);
      if (s > bps) { bps = s; bp = { type: p.type, lot: p.lot, raw: tx }; }
    }
    let bh = { bacteria: '', no: '', dilute: '', weak: false, raw: '' }, bhs = -1;
    for (const tx of await read(prepRegion(panelCanvas, 0, 0, 0.48, band), ['11', '6'])) {
      const p = parseHand(tx);
      const s = (p.bacteria ? 1 : 0) + (p.weak ? -0.5 : 0) + (p.no ? 1 : 0) + (p.dilute ? 1 : 0)
              + (/\d\s*[-/]\s*\d/.test(tx) ? 1 : 0);
      if (s > bhs) { bhs = s; bh = { ...p, raw: tx }; }
    }
    return {
      type: bp.type, lot: bp.lot, bacteria: bh.bacteria, no: bh.no, dilute: bh.dilute,
      weak: { bacteria: bh.weak }, rawPrinted: bp.raw, rawHandwriting: bh.raw,
    };
  } catch (error) {
    console.warn('OCR 처리 실패:', error);
    return { type: '', lot: '', bacteria: '', no: '', dilute: '',
      weak: { bacteria: false }, rawHandwriting: '(인식 실패)', rawPrinted: '' };
  }
}

function guessMetaFromFilename(name) {
  const up = String(name || '').toUpperCase();
  const t = up.match(/(?:^|[^A-Z])(AC|EC|YM|SA|CC)(?![A-Z])/);
  const bac = up.match(/(?:^|[^A-Z0-9])([A-Z]{2,4})[ _-]?(\d)[ _-]?(\d)(?![0-9])/);
  return { type: t ? t[1] : '', bacteria: bac ? bac[1] : '', no: bac ? bac[2] : '', dilute: bac ? bac[3] : '' };
}

function resolveOcrFields(ocr, guess, prevRecord) {
  const fields = ['type', 'lot', 'bacteria', 'no', 'dilute'];
  const inheritable = new Set(['type', 'lot', 'bacteria']);
  const weak = (ocr && ocr.weak) || {};
  const values = {}, source = {}, needsReview = {};
  for (const f of fields) {
    let v = '', src = '';
    if (ocr && ocr[f]) { v = ocr[f]; src = 'ocr'; }
    else if (guess && guess[f]) { v = guess[f]; src = 'file'; }
    else if (inheritable.has(f) && prevRecord && prevRecord[f]) { v = prevRecord[f]; src = 'prev'; }
    values[f] = v; source[f] = src;
    needsReview[f] = !v || !!weak[f] || src !== 'ocr';
  }
  return { values, source, needsReview };
}

  return new Promise(res => {
    const s = document.createElement('script');

async function hasVendor(file) {

  const p = e.payload || {};
  switch (e.kind) {
    case 'PLATE_ADD': return `#${p.id} 추가 · ${p.filename} · CFU ${p.count ?? '-'}`;
    case 'COUNT_SET': return `#${p.id} 콜로니 ${p.from} → ${p.to} (${p.how || ''})`;

    return {
      ...rr, canvas, lowRes: true, rotation: 0,
      autoColonies: JSON.parse(JSON.stringify(rr.colonies || [])),
      historyPast: [], historyFuture: [], selected: false,
      sensitivity: DEFAULTS.sensitivity, minArea: DEFAULTS.minArea,
      colorMode: DEFAULTS.colorMode, labelHeightRatio: DEFAULTS.labelHeightRatio,
      needsReview: rr.needsReview || {}, ocrSource: rr.ocrSource || {},
      rawHandwriting: '', rawPrinted: '',
    };
  }));
  state.records = recs;
  state.events = r.events;
  ledger.commit('IMPORT_ACK', { events: r.events.length, verifiedTip: ledger.tip(r.events) });
  saveState();
  refreshTable();
  renderLedgerModal();
}
```
